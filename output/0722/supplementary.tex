\subsection{Region-Level Read-Before-Write Training Procedure}
Algorithm~\ref{alg:rc_memnet_verified} presents the exact region-level read-before-write training and adaptation procedure implemented in \texttt{RCDMNet}.

\begin{algorithm}[h]
\caption{Region-Level Read-Before-Write Training and Adaptation}
\label{alg:rc_memnet_verified}
\begin{algorithmic}[1]
\REQUIRE Source datasets $\mathcal{D}_1, \dots, \mathcal{D}_S$; Target support set $\mathcal{D}_{target}^{supp}$; Max steps $T_{src}=3000$; Adaptation steps $T_{tgt}=30$.
\STATE Initialize $\theta$ (Encoder, RPL, RGMA Keys $K$), $K \sim \mathcal{N}(0, 0.02)$, $V \sim \mathcal{N}(0, 0.02)$.
\STATE Initialize cumulative memory states $S_{c,j} \leftarrow 0, N_{c,j} \leftarrow 0$.
\COMMENT{\textbf{Source Pre-training Phase (3,000 Steps)}}
\FOR{each epoch}
    \STATE Randomly permute source region order $\pi = \text{Shuffle}(1, \dots, S)$.
    \FOR{each region $r \in \pi$}
        \STATE \textbf{Pass 1: Predictive Optimization (Differentiable)}
        \FOR{each batch $(x_i^r, y_i^r) \in \mathcal{D}_r$}
            \STATE Extract acoustic features $h_i^r$ and regional context $c_i^r$.
            \STATE Fuse RPL output: $u_i^r = F_p([h_i^r \Vert c_i^r])$.
            \STATE Modulate memory query: $q_i^r = \text{LN}((1+\tanh(\gamma(c_i^r))) \odot W_q h_i^r + \beta(c_i^r))$.
            \STATE Read memory: $A_i^r = \text{softmax}(\bar q_i^r \bar K^\top / \tau_a)$, $m_i^r = A_i^r V$.
            \STATE Fuse final representation: $\widetilde h_i^r = F_\omega([u_i^r \Vert m_i^r])$.
            \STATE Compute logits $o_i^r = C_\psi(\widetilde h_i^r)$ and update $\theta$ via $\text{CE}(o_i^r, y_i^r)$.
        \ENDFOR
        \STATE \textbf{Pass 2: Region-Level Writing Pass (No-Gradient)}
        \FOR{each sample $i \in \mathcal{D}_r$}
            \STATE Extract write feature $z_i^r = \text{sg}[\text{Norm}(W_v h_i^r)]$.
            \STATE Compute Top-1 sparse affinity $\widetilde e_{i,c,j}^r$ within ground-truth class $y_i^r = c$.
        \ENDFOR
        \STATE Aggregate regional candidate $\widehat z_{r,c,j}$ and normalized weight $\omega_{r,c,j}$.
        \STATE Update cumulative states: $S_{c,j} \leftarrow S_{c,j} + \omega_{r,c,j}\widehat z_{r,c,j}$, $N_{c,j} \leftarrow N_{c,j} + \omega_{r,c,j}$.
        \STATE Consolidate Memory Values: $v_{c,j} \leftarrow \text{Norm}(S_{c,j} / (N_{c,j} + \epsilon))$.
    \ENDFOR
\ENDFOR
\STATE Freeze network parameters $\theta$ and Memory Values $V$.
\COMMENT{\textbf{Target Few-Shot Adaptation Phase (30 Steps)}}
\STATE Initialize target composition weights $\boldsymbol{\lambda}_t = [1/S, \dots, 1/S]$.
\FOR{$step = 1$ \TO $T_{tgt}$}
    \STATE Construct target base prompt: $p_t = \sum_{r=1}^S \lambda_{t,r} p_r$.
    \STATE Evaluate loss $\mathcal{L}_{supp}$ on 4-shot support set $\mathcal{D}_{target}^{supp}$.
    \STATE Update only composition weights $\boldsymbol{\lambda}_t$ via AdamW ($lr=10^{-3}$).
\ENDFOR
\RETURN Target predictions on query set $\mathcal{D}_{target}^{query}$.
\end{algorithmic}
\end{algorithm}
